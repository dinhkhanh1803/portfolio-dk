```ts
export const WORLD_WIDTH = 1280;
export const WORLD_HEIGHT = 720;
export const FIXED_STEP_MS = 16;
export const BALL_RADIUS = 12;

export type MatchMode = "ai" | "local";
export type Difficulty = "easy" | "normal" | "hard";
export type MatchPhase = "ready" | "playing" | "paused" | "gameover";

export type PongInput = {
  leftUp: boolean;
  leftDown: boolean;
  rightUp: boolean;
  rightDown: boolean;
};

export type PongState = {
  phase: MatchPhase;
  mode: MatchMode;
  difficulty: Difficulty;
  targetScore: 5 | 7 | 11;
  leftScore: number;
  rightScore: number;
  winner: "left" | "right" | null;
  leftPaddle: { x: number; y: number; width: number; height: number; speed: number };
  rightPaddle: { x: number; y: number; width: number; height: number; speed: number };
  ball: { x: number; y: number; vx: number; vy: number; radius: number; speed: number };
  rallyHits: number;
  elapsedMs: number;
  serveDirection: -1 | 1;
};

const clamp = (value: number, min: number, max: number) =>
  Math.max(min, Math.min(max, value));

export function createMatch(settings: {
  mode: MatchMode;
  targetScore: 5 | 7 | 11;
  difficulty: Difficulty;
}): PongState {
  const paddle = { width: 22, height: 128, speed: 570 };
  return {
    phase: "ready",
    ...settings,
    leftScore: 0,
    rightScore: 0,
    winner: null,
    leftPaddle: { ...paddle, x: 54, y: 296 },
    rightPaddle: { ...paddle, x: 1204, y: 296 },
    ball: { x: 640, y: 360, vx: 0, vy: 0, radius: BALL_RADIUS, speed: 430 },
    rallyHits: 0,
    elapsedMs: 0,
    serveDirection: 1,
  };
}

export function startRally(state: PongState): PongState {
  if (state.phase !== "ready") return state;
  const angle = 0.24;
  return {
    ...state,
    phase: "playing",
    ball: {
      ...state.ball,
      vx: Math.cos(angle) * state.ball.speed * state.serveDirection,
      vy: Math.sin(angle) * state.ball.speed,
    },
  };
}

export function step(state: PongState, deltaMs: number, input: PongInput): PongState {
  if (state.phase !== "playing") return state;
  const dt = Math.max(0, deltaMs) / 1000;
  const move = (up: boolean, down: boolean) => (Number(down) - Number(up)) * 570 * dt;
  return {
    ...state,
    elapsedMs: state.elapsedMs + deltaMs,
    leftPaddle: {
      ...state.leftPaddle,
      y: clamp(state.leftPaddle.y + move(input.leftUp, input.leftDown), 0, WORLD_HEIGHT - state.leftPaddle.height),
    },
    rightPaddle: {
      ...state.rightPaddle,
      y: clamp(state.rightPaddle.y + move(input.rightUp, input.rightDown), 0, WORLD_HEIGHT - state.rightPaddle.height),
    },
    ball: {
      ...state.ball,
      x: state.ball.x + state.ball.vx * dt,
      y: state.ball.y + state.ball.vy * dt,
    },
  };
}

export function advanceFixed(
  state: PongState,
  accumulatorMs: number,
  deltaMs: number,
  input: PongInput,
) {
  let next = state;
  let remaining = Math.max(0, accumulatorMs + Math.max(0, deltaMs));
  while (remaining >= FIXED_STEP_MS && next.phase === "playing") {
    next = step(next, FIXED_STEP_MS, input);
    remaining -= FIXED_STEP_MS;
  }
  return { state: next, accumulatorMs: next.phase === "playing" ? remaining : 0 };
}
```

```ts
const BASE_BALL_SPEED = 430;
const MAX_BALL_SPEED = 980;
const HIT_ACCELERATION = 1.055;
const MAX_REBOUND_ANGLE = Math.PI * 0.34;

function resetBall(state: PongState, serveDirection: -1 | 1): PongState {
  return {
    ...state,
    phase: "ready",
    serveDirection,
    rallyHits: 0,
    ball: {
      x: WORLD_WIDTH / 2,
      y: WORLD_HEIGHT / 2,
      vx: 0,
      vy: 0,
      radius: BALL_RADIUS,
      speed: BASE_BALL_SPEED,
    },
  };
}

function awardPoint(state: PongState, side: "left" | "right"): PongState {
  const leftScore = state.leftScore + Number(side === "left");
  const rightScore = state.rightScore + Number(side === "right");
  const winner = leftScore >= state.targetScore ? "left" : rightScore >= state.targetScore ? "right" : null;
  const reset = resetBall({ ...state, leftScore, rightScore, winner }, side === "left" ? 1 : -1);
  return winner ? { ...reset, phase: "gameover", winner } : reset;
}
```

Update `step()` so it:

- Calculates previous and next ball edges.
- Reflects from top/bottom walls.
- Detects crossing of a paddle face during the step.
- Places the ball outside the paddle after collision.
- Computes `relativeHit = clamp((ballY - paddleCenter) / (paddleHeight / 2), -1, 1)`.
- Uses `relativeHit * MAX_REBOUND_ANGLE` for vertical velocity.
- Accelerates by `HIT_ACCELERATION` up to `MAX_BALL_SPEED`.
- Calls `awardPoint()` when the ball fully exits left or right.

```ts
const AI_CONFIG = {
  easy: { maxSpeed: 330, reactionMs: 210, error: 92 },
  normal: { maxSpeed: 465, reactionMs: 120, error: 44 },
  hard: { maxSpeed: 590, reactionMs: 68, error: 16 },
} as const;

export const aiConfig = (difficulty: Difficulty) => AI_CONFIG[difficulty];

export function updateAiTarget(state: PongState, randomValue: number): PongState {
  if (state.mode !== "ai") return state;
  const config = aiConfig(state.difficulty);
  const travelSeconds = state.ball.vx > 0
    ? Math.max(0, (state.rightPaddle.x - state.ball.x) / state.ball.vx)
    : 0;
  const projected = state.ball.y + state.ball.vy * travelSeconds;
  const error = (Math.max(0, Math.min(1, randomValue)) * 2 - 1) * config.error;
  return { ...state, aiTargetY: clamp(projected + error, 0, WORLD_HEIGHT) };
}
```

```ts
export class PongAudio {
  private context: AudioContext | null = null;
  private muted = false;

  async unlock() {
    if (this.muted || typeof window === "undefined") return;
    try {
      this.context ??= new window.AudioContext();
      if (this.context.state === "suspended") await this.context.resume();
    } catch {
      this.context = null;
    }
  }

  setMuted(muted: boolean) {
    this.muted = muted;
    if (muted && this.context?.state === "running") {
      void this.context.suspend().catch(() => undefined);
    }
  }

  private tone(frequency: number, duration: number, gain: number, type: OscillatorType) {
    const context = this.context;
    if (!context || this.muted || context.state !== "running") return;
    try {
      const oscillator = context.createOscillator();
      const envelope = context.createGain();
      const start = context.currentTime;
      const end = start + duration;
      oscillator.type = type;
      oscillator.frequency.setValueAtTime(frequency, start);
      envelope.gain.setValueAtTime(0.0001, start);
      envelope.gain.exponentialRampToValueAtTime(gain, start + 0.008);
      envelope.gain.exponentialRampToValueAtTime(0.0001, end);
      oscillator.connect(envelope);
      envelope.connect(context.destination);
      oscillator.start(start);
      oscillator.stop(end + 0.01);
    } catch {}
  }

  playPaddleHit() { this.tone(540, 0.06, 0.045, "square"); }
  playWallBounce() { this.tone(310, 0.045, 0.03, "triangle"); }
  playScore() { this.tone(190, 0.18, 0.055, "sawtooth"); }
  playCountdown() { this.tone(440, 0.08, 0.035, "sine"); }
  playVictory() { this.tone(760, 0.32, 0.06, "triangle"); }

  dispose() {
    const context = this.context;
    this.context = null;
    if (context) void context.close().catch(() => undefined);
  }
}
```
